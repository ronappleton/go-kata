package kata import "strconv" // FizzBuzz returns a slice of strings for numbers 1 through n. // Multiples of 3 become "Fizz", 5 become "Buzz", both become "FizzBuzz". func FizzBuzz(n int) []string { if n <= 0 { return []string{} } result := make([]string, n) for i := 1; i <= n; i++ { switch { case i%15 == 0: result[i-1] = "FizzBuzz" case i%3 == 0: result[i-1] = "Fizz" case i%5 == 0: result[i-1] = "Buzz" default: result[i-1] = strconv.Itoa(i) } } return result }
