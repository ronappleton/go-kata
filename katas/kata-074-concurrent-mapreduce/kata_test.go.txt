package kata

import (
	"testing"
)

func TestMapReduce(t *testing.T) {
	t.Run("map reduce sum of squares", func(t *testing.T) {
		got, err := MapReduce([]int{1, 2, 3, 4}, 4,
			func(x int) int { return x * x },
			func(rs []int) int { s := 0; for _, r := range rs { s += r }; return s })
		if err != nil {
			t.Fatal(err)
		}
		if got != 30 {
			t.Errorf("got %d, want 30", got)
		}
	})
	t.Run("empty items returns zero value", func(t *testing.T) {
		got, err := MapReduce([]int{}, 4,
			func(x int) int { return x },
			func(rs []int) int { return len(rs) })
		if err != nil {
			t.Fatal(err)
		}
		if got != 0 {
			t.Errorf("got %d, want 0", got)
		}
	})
	t.Run("string concat reduce", func(t *testing.T) {
		got, err := MapReduce([]string{"a", "b", "c"}, 2,
			func(s string) string { return s + "!" },
			func(rs []string) string { out := ""; for _, r := range rs { out += r }; return out })
		if err != nil {
			t.Fatal(err)
		}
		if got != "a!b!c!" {
			t.Errorf("got %q, want a!b!c!", got)
		}
	})
	t.Run("workers below 1 errors", func(t *testing.T) {
		_, err := MapReduce([]int{1}, 0,
			func(x int) int { return x },
			func(rs []int) int { return 0 })
		if err == nil {
			t.Error("expected error for workers < 1")
		}
	})
}
