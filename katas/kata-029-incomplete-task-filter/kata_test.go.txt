package kata

import (
	"reflect"
	"testing"
)

func TestIncompleteTasks(t *testing.T) {
	t.Run("input not mutated", func(t *testing.T) {
		in := []Task{{Title: "a", Done: false}, {Title: "b", Done: true}}
		_ = IncompleteTasks(in)
		if len(in) != 2 || in[0].Done {
			t.Errorf("IncompleteTasks mutated input: %v", in)
		}
	})
	tests := []struct {
		name string
		in   []Task
		want []Task
	}{
		{"nil", nil, []Task{}},
		{"all done", []Task{{"a", true}, {"b", true}}, []Task{}},
		{"none done", []Task{{"a", false}, {"b", false}}, []Task{{"a", false}, {"b", false}}},
		{"mixed preserves order", []Task{{"a", true}, {"b", false}, {"c", true}, {"d", false}}, []Task{{"b", false}, {"d", false}}},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got := IncompleteTasks(tt.in)
			if !reflect.DeepEqual(got, tt.want) {
				t.Errorf("IncompleteTasks(%v) = %v, want %v", tt.in, got, tt.want)
			}
		})
	}
}
