package kata

import (
	"context"
	"net/http"
	"net/http/httptest"
	"sync/atomic"
	"testing"
)

func TestScrapeCollectsBodies(t *testing.T) {
	var hits atomic.Int32
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		hits.Add(1)
		switch r.URL.Path {
		case "/a":
			w.Write([]byte("AAA"))
		case "/b":
			w.Write([]byte("BBB"))
		default:
			w.WriteHeader(404)
		}
	}))
	defer srv.Close()

	got, err := Scrape(context.Background(), []string{srv.URL + "/a", srv.URL + "/b"}, 1000, srv.Client())
	if err != nil {
		t.Fatalf("unexpected error: %v", err)
	}
	if got[srv.URL+"/a"] != "AAA" || got[srv.URL+"/b"] != "BBB" {
		t.Errorf("got = %v", got)
	}
	if hits.Load() != 2 {
		t.Errorf("hits = %d, want 2", hits.Load())
	}
}

func TestScrapeReturnsErrorForNon2xx(t *testing.T) {
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.WriteHeader(500)
	}))
	defer srv.Close()

	got, err := Scrape(context.Background(), []string{srv.URL + "/x"}, 1000, srv.Client())
	if err == nil {
		t.Fatal("expected error for 500 response")
	}
	if len(got) != 0 {
		t.Errorf("partial map should be empty, got %v", got)
	}
}

func TestScrapeCancellation(t *testing.T) {
	srv := httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
		w.Write([]byte("ok"))
	}))
	defer srv.Close()

	ctx, cancel := context.WithCancel(context.Background())
	cancel()
	_, err := Scrape(ctx, []string{srv.URL + "/a"}, 1000, srv.Client())
	if err == nil {
		t.Error("expected error when context is already cancelled")
	}
}
