package kata

import (
	"testing"

func TestFixStuttering(t *testing.T) {
	tests := []struct {
		name, pkg, want string
	}{
	{"HTTPClient", "http", "Client"},
	{"UserRecord", "user", "Record"},
	{"ServerConfig", "server", "Config"},
	{"Config", "server", "Config"},    // no stuttering
	{"Client", "http", "Client"},      // no stuttering
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got := FixStuttering(tt.name, tt.pkg)
			if got != tt.want {
				t.Errorf("FixStuttering(%q, %q) = %q, want %q", tt.name, tt.pkg, got, tt.want)
			}
		})
	}
}

func TestValidateReceiverNames(t *testing.T) {
	good := []string{"u.Name", "u.Email", "u.Save"}
	if bad := ValidateReceiverNames(good); len(bad) != 0 {
		t.Errorf("ValidateReceiverNames(%v) = %v, want []", good, bad)
	}

	bad := []string{"u.Name", "user.Email", "u.Save"}
	if bad2 := ValidateReceiverNames(bad); len(bad2) == 0 {
		t.Errorf("ValidateReceiverNames(%v) = [], want issues", bad)
	}
}
