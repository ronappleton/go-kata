package kata

import (
	"testing"
)

func TestShortestPathSimple(t *testing.T) {
	edges := map[string]map[string]int{
		"A": {"B": 1, "C": 4},
		"B": {"C": 1, "D": 5},
		"C": {"D": 1},
		"D": {},
	}
	cost, path, err := ShortestPath(edges, "A", "D")
	if err != nil {
		t.Fatal(err)
	}
	// A -> B -> C -> D = 1+1+1 = 3 beats A -> C -> D = 4+1 = 5
	if cost != 3 {
		t.Errorf("cost = %d, want 3", cost)
	}
	if len(path) == 0 || path[0] != "A" || path[len(path)-1] != "D" {
		t.Errorf("path = %v, want A...D", path)
	}
	// verify the path is real and sums to cost
	sum := 0
	for i := 0; i < len(path)-1; i++ {
		w, ok := edges[path[i]][path[i+1]]
		if !ok {
			t.Fatalf("path edge %s->%s does not exist", path[i], path[i+1])
		}
		sum += w
	}
	if sum != cost {
		t.Errorf("path sum = %d, want %d", sum, cost)
	}
}

func TestShortestPathStartEqualsGoal(t *testing.T) {
	edges := map[string]map[string]int{"A": {"B": 2}, "B": {}}
	cost, path, err := ShortestPath(edges, "A", "A")
	if err != nil {
		t.Fatal(err)
	}
	if cost != 0 {
		t.Errorf("cost = %d, want 0", cost)
	}
	if len(path) != 1 || path[0] != "A" {
		t.Errorf("path = %v, want [A]", path)
	}
}

func TestShortestPathUnreachable(t *testing.T) {
	edges := map[string]map[string]int{
		"A": {"B": 1},
		"B": {},
		"C": {"D": 1},
		"D": {},
	}
	if _, _, err := ShortestPath(edges, "A", "C"); err == nil {
		t.Error("expected error for unreachable goal")
	}
}

func TestShortestPathDisconnectedNode(t *testing.T) {
	edges := map[string]map[string]int{
		"A": {},
		"B": {"A": 3},
	}
	// B can reach A; A cannot reach B
	if _, _, err := ShortestPath(edges, "A", "B"); err == nil {
		t.Error("expected error when start cannot reach goal")
	}
}
