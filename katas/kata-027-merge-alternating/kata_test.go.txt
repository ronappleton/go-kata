package kata

import (
	"reflect"
	"testing"
)

func TestMergeAlternating(t *testing.T) {
	tests := []struct {
		name string
		a, b []int
		want []int
	}{
		{"equal lengths", []int{1, 3, 5}, []int{2, 4, 6}, []int{1, 2, 3, 4, 5, 6}},
		{"a longer", []int{1, 3, 5, 7}, []int{2, 4}, []int{1, 2, 3, 4, 5, 7}},
		{"b longer", []int{1, 3}, []int{2, 4, 6, 8}, []int{1, 2, 3, 4, 6, 8}},
		{"a nil", nil, []int{1, 2}, []int{1, 2}},
		{"b nil", []int{1, 2}, nil, []int{1, 2}},
		{"both nil", nil, nil, []int{}},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got := MergeAlternating(tt.a, tt.b)
			if !reflect.DeepEqual(got, tt.want) {
				t.Errorf("MergeAlternating(%v, %v) = %v, want %v", tt.a, tt.b, got, tt.want)
			}
		})
	}
}
