package kata

import (
	"errors"
	"testing"
)

type engine struct {
	id int
}

type car struct {
	e *engine
}

func TestContainerRegisterResolve(t *testing.T) {
	c := NewContainer()
	if err := c.Register("engine", func() any { return &engine{id: 1} }); err != nil {
		t.Fatal(err)
	}
	v, err := c.Resolve("engine")
	if err != nil {
		t.Fatal(err)
	}
	e, ok := v.(*engine)
	if !ok {
		t.Fatalf("resolved %T, want *engine", v)
	}
	if e.id != 1 {
		t.Errorf("engine id = %d, want 1", e.id)
	}
}

func TestContainerSingleton(t *testing.T) {
	c := NewContainer()
	calls := 0
	if err := c.Register("svc", func() any {
		calls++
		return &car{}
	}); err != nil {
		t.Fatal(err)
	}
	a, _ := c.Resolve("svc")
	b, _ := c.Resolve("svc")
	if a != b {
		t.Error("Resolve should return the same singleton instance")
	}
	if calls != 1 {
		t.Errorf("constructor called %d times, want 1", calls)
	}
}

func TestContainerResolveUnregistered(t *testing.T) {
	c := NewContainer()
	if _, err := c.Resolve("missing"); err == nil {
		t.Error("expected error for unregistered name")
	}
}

func TestContainerDuplicateRegister(t *testing.T) {
	c := NewContainer()
	c.Register("x", func() any { return 1 })
	if err := c.Register("x", func() any { return 2 }); err == nil {
		t.Error("expected error for duplicate registration")
	}
}

func TestContainerNoPanicOnConstructorError(t *testing.T) {
	c := NewContainer()
	if err := c.Register("boom", func() any { panic("ctor panicked") }); err != nil {
		t.Fatal(err)
	}
	defer func() {
		if r := recover(); r != nil {
			t.Errorf("Resolve should not panic, recovered %v", r)
		}
	}()
	c.Resolve("boom")
}

var _ = errors.New
