package kata

import "testing"

func TestStatusClass(t *testing.T) {
	tests := []struct {
		name string
		in   int
		want string
	}{
		{"1xx", 100, "informational"},
		{"1xx 199", 199, "informational"},
		{"2xx", 200, "success"},
		{"2xx 299", 299, "success"},
		{"3xx", 301, "redirect"},
		{"4xx", 404, "client"},
		{"5xx", 500, "server"},
		{"5xx 599", 599, "server"},
		{"unknown low", 99, "unknown"},
		{"unknown high", 600, "unknown"},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			if got := StatusClass(tt.in); got != tt.want {
				t.Errorf("StatusClass(%d) = %q, want %q", tt.in, got, tt.want)
			}
		})
	}
}
