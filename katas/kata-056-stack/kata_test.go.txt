package kata

import "testing"

func TestStack(t *testing.T) {
	t.Run("empty stack", func(t *testing.T) {
		var s Stack[int]
		if s.Len() != 0 {
			t.Errorf("Len() = %d, want 0", s.Len())
		}
		if _, ok := s.Pop(); ok {
			t.Error("Pop on empty should be ok=false")
		}
		if _, ok := s.Peek(); ok {
			t.Error("Peek on empty should be ok=false")
		}
	})
	t.Run("push pop LIFO", func(t *testing.T) {
		var s Stack[string]
		s.Push("a")
		s.Push("b")
		if v, ok := s.Pop(); !ok || v != "b" {
			t.Errorf("Pop = (%q, %v), want b, true", v, ok)
		}
		if v, ok := s.Pop(); !ok || v != "a" {
			t.Errorf("Pop = (%q, %v), want a, true", v, ok)
		}
		if _, ok := s.Pop(); ok {
			t.Error("Pop after empty should fail")
		}
	})
	t.Run("peek does not remove", func(t *testing.T) {
		var s Stack[int]
		s.Push(42)
		if v, ok := s.Peek(); !ok || v != 42 {
			t.Errorf("Peek = (%d, %v), want (42, true)", v, ok)
		}
		if s.Len() != 1 {
			t.Errorf("Len after Peek = %d, want 1", s.Len())
		}
	})
}
