package kata

import (
	"context"
	"reflect"
	"testing"
)

func TestPipeline(t *testing.T) {
	double := func(ctx context.Context, in <-chan int) <-chan int {
		out := make(chan int)
		go func() {
			defer close(out)
			for v := range in {
				out <- v * 2
			}
		}()
		return out
	}
	addOne := func(ctx context.Context, in <-chan int) <-chan int {
		out := make(chan int)
		go func() {
			defer close(out)
			for v := range in {
				out <- v + 1
			}
		}()
		return out
	}
	t.Run("composes stages in order", func(t *testing.T) {
		in := make(chan int)
		out := Pipeline(context.Background(), in, double, addOne)
		go func() {
			for i := 1; i <= 3; i++ {
				in <- i
			}
			close(in)
		}()
		var got []int
		for v := range out {
			got = append(got, v)
		}
		want := []int{3, 5, 7} // (1*2)+1, (2*2)+1, (3*2)+1
		if !reflect.DeepEqual(got, want) {
			t.Errorf("got %v, want %v", got, want)
		}
	})
}
