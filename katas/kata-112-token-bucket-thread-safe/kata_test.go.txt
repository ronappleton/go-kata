package kata

import (
	"sync"
	"testing"
	"time"
)

func TestNewTokenBucket(t *testing.T) {
	if _, err := NewTokenBucket(0, 1); err == nil {
		t.Error("expected error for ratePerSec 0")
	}
	if _, err := NewTokenBucket(1, 0); err == nil {
		t.Error("expected error for burst 0")
	}
	if _, err := NewTokenBucket(-2, 3); err == nil {
		t.Error("expected error for negative rate")
	}
	if _, err := NewTokenBucket(1, 1); err != nil {
		t.Errorf("unexpected error: %v", err)
	}
}

func TestTokenBucketBurst(t *testing.T) {
	b, _ := NewTokenBucket(1000, 3) // fast refill; burst dominates at t=0
	allowed := 0
	for i := 0; i < 10; i++ {
		if b.Allow() {
			allowed++
		}
	}
	if allowed < 3 {
		t.Errorf("allowed = %d, want at least burst(3)", allowed)
	}
}

func TestTokenBucketRefill(t *testing.T) {
	b, _ := NewTokenBucket(100, 1) // 100/s => 1 token per 10ms
	if !b.Allow() {
		t.Fatal("first Allow should succeed")
	}
	if b.Allow() {
		t.Error("second immediate Allow should fail (burst=1, no refill yet)")
	}
	time.Sleep(25 * time.Millisecond) // ~2 tokens refilled, burst caps at 1
	if !b.Allow() {
		t.Error("Allow after refill should succeed")
	}
}

func TestTokenBucketConcurrent(t *testing.T) {
	b, _ := NewTokenBucket(100000, 1000)
	var wg sync.WaitGroup
	for i := 0; i < 8; i++ {
		wg.Add(1)
		go func() {
			defer wg.Done()
			for j := 0; j < 100; j++ {
				b.Allow()
			}
		}()
	}
	wg.Wait()
}
