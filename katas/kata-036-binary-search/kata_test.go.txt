package kata

import "testing"

func TestBinarySearch(t *testing.T) {
	tests := []struct {
		name   string
		sorted []int
		target int
		want   int
	}{
		{"empty", []int{}, 5, -1},
		{"nil", nil, 5, -1},
		{"found first", []int{1, 3, 5, 7, 9}, 1, 0},
		{"found middle", []int{1, 3, 5, 7, 9}, 5, 2},
		{"found last", []int{1, 3, 5, 7, 9}, 9, 4},
		{"not found low", []int{1, 3, 5, 7, 9}, 0, -1},
		{"not found between", []int{1, 3, 5, 7, 9}, 4, -1},
		{"not found high", []int{1, 3, 5, 7, 9}, 10, -1},
		{"single found", []int{7}, 7, 0},
		{"single not found", []int{7}, 8, -1},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			if got := BinarySearch(tt.sorted, tt.target); got != tt.want {
				t.Errorf("BinarySearch(%v, %d) = %d, want %d", tt.sorted, tt.target, got, tt.want)
			}
		})
	}
}
