package kata

import (
	"encoding/base64"
	"strings"
	"testing"
	"time"
)

func TestSignJWTStructure(t *testing.T) {
	token, err := SignJWT(map[string]any{"sub": "user-1"}, []byte("secret"), time.Hour)
	if err != nil {
		t.Fatal(err)
	}
	parts := strings.Split(token, ".")
	if len(parts) != 3 {
		t.Fatalf("token = %q, want 3 dot-separated parts", token)
	}
	// header decodes to {"alg":"HS256","typ":"JWT"}
	hdr, err := base64.RawURLEncoding.DecodeString(parts[0])
	if err != nil {
		t.Fatal(err)
	}
	if !strings.Contains(string(hdr), "HS256") {
		t.Errorf("header = %s, want HS256", hdr)
	}
}

func TestVerifyJWTReturnsClaims(t *testing.T) {
	token, err := SignJWT(map[string]any{"sub": "user-1", "role": "admin"}, []byte("secret"), time.Hour)
	if err != nil {
		t.Fatal(err)
	}
	claims, err := VerifyJWT(token, []byte("secret"))
	if err != nil {
		t.Fatal(err)
	}
	if claims["sub"] != "user-1" || claims["role"] != "admin" {
		t.Errorf("claims = %v", claims)
	}
}

func TestVerifyJWTExpClaimPresent(t *testing.T) {
	token, _ := SignJWT(map[string]any{}, []byte("secret"), time.Hour)
	claims, err := VerifyJWT(token, []byte("secret"))
	if err != nil {
		t.Fatal(err)
	}
	exp, ok := claims["exp"].(float64)
	if !ok {
		t.Fatalf("exp claim missing or wrong type: %v", claims["exp"])
	}
	if exp <= float64(time.Now().Unix()) {
		t.Errorf("exp = %v, want in the future", exp)
	}
}

func TestVerifyJWTWrongSecret(t *testing.T) {
	token, _ := SignJWT(map[string]any{"sub": "x"}, []byte("right"), time.Hour)
	if _, err := VerifyJWT(token, []byte("wrong")); err == nil {
		t.Error("expected error for wrong secret")
	}
}

func TestVerifyJWTTampered(t *testing.T) {
	token, _ := SignJWT(map[string]any{"sub": "x"}, []byte("secret"), time.Hour)
	parts := strings.Split(token, ".")
	payload, _ := base64.RawURLEncoding.DecodeString(parts[1])
	payload = []byte(strings.Replace(string(payload), "x", "y", 1))
	parts[1] = base64.RawURLEncoding.EncodeToString(payload)
	tampered := strings.Join(parts, ".")
	if _, err := VerifyJWT(tampered, []byte("secret")); err == nil {
		t.Error("expected error for tampered payload")
	}
}

func TestVerifyJWTExpired(t *testing.T) {
	token, _ := SignJWT(map[string]any{}, []byte("secret"), -time.Hour)
	if _, err := VerifyJWT(token, []byte("secret")); err == nil {
		t.Error("expected error for expired token")
	}
}

func TestVerifyJWTMalformed(t *testing.T) {
	if _, err := VerifyJWT("not.a.jwt", []byte("secret")); err == nil {
		t.Error("expected error for malformed token")
	}
}
