package kata

import (
	"reflect"
	"testing"
)

func TestNormalizeTags(t *testing.T) {
	tests := []struct {
		name string
		in   string
		want []string
	}{
		{"simple", "go, testing", []string{"go", "testing"}},
		{"trims spaces", "  go  ,  testing ", []string{"go", "testing"}},
		{"lowercases", "Go, TESTING", []string{"go", "testing"}},
		{"drops empties", "go,,testing,,", []string{"go", "testing"}},
		{"empty input", "", []string{}},
		{"only commas", ",,,", []string{}},
		{"preserves order", "b, a, c", []string{"b", "a", "c"}},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got := NormalizeTags(tt.in)
			if !reflect.DeepEqual(got, tt.want) {
				t.Errorf("NormalizeTags(%q) = %v, want %v", tt.in, got, tt.want)
			}
		})
	}
}
