package kata

import (
	"reflect"
	"testing"
)

func TestFilterEven(t *testing.T) {
	t.Run("input not mutated", func(t *testing.T) {
		in := []int{1, 2, 3, 4}
		_ = FilterEven(in)
		if !reflect.DeepEqual(in, []int{1, 2, 3, 4}) {
			t.Errorf("FilterEven mutated input: %v", in)
		}
	})
	tests := []struct {
		name string
		in   []int
		want []int
	}{
		{"nil", nil, []int{}},
		{"empty", []int{}, []int{}},
		{"all even", []int{2, 4, 6}, []int{2, 4, 6}},
		{"all odd", []int{1, 3, 5}, []int{}},
		{"mixed", []int{1, 2, 3, 4, 5, 6}, []int{2, 4, 6}},
		{"zero is even", []int{0, 1}, []int{0}},
		{"negatives", []int{-2, -3, -4}, []int{-2, -4}},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got := FilterEven(tt.in)
			if !reflect.DeepEqual(got, tt.want) {
				t.Errorf("FilterEven(%v) = %v, want %v", tt.in, got, tt.want)
			}
		})
	}
}
