package kata

import "testing"

func TestTokenBucket(t *testing.T) {
	t.Run("invalid params error", func(t *testing.T) {
		if _, err := NewTokenBucket(0, 1); err == nil {
			t.Error("ratePerSec=0 should error")
		}
		if _, err := NewTokenBucket(1, 0); err == nil {
			t.Error("burst=0 should error")
		}
	})
	t.Run("burst limits immediate allowance", func(t *testing.T) {
		tb, err := NewTokenBucket(100, 3)
		if err != nil {
			t.Fatal(err)
		}
		allowed := 0
		for i := 0; i < 10; i++ {
			if tb.Allow() {
				allowed++
			}
		}
		if allowed > 3 {
			t.Errorf("burst=3 should cap immediate allows, got %d", allowed)
		}
	})
}
