package kata

import "testing"

func TestMinMax(t *testing.T) {
	t.Run("empty slice errors", func(t *testing.T) {
		if _, _, err := MinMax(nil); err == nil {
			t.Error("MinMax(nil) expected an error")
		}
		if _, _, err := MinMax([]int{}); err == nil {
			t.Error("MinMax([]) expected an error")
		}
	})
	tests := []struct {
		name    string
		in      []int
		wantMin int
		wantMax int
	}{
		{"single element", []int{5}, 5, 5},
		{"ascending", []int{1, 2, 3, 4}, 1, 4},
		{"descending", []int{9, 8, 7}, 7, 9},
		{"negatives", []int{-5, -1, -9}, -9, -1},
		{"mixed", []int{3, -2, 10, 0}, -2, 10},
		{"duplicates", []int{4, 4, 4}, 4, 4},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			gotMin, gotMax, err := MinMax(tt.in)
			if err != nil {
				t.Fatalf("MinMax(%v) unexpected error: %v", tt.in, err)
			}
			if gotMin != tt.wantMin || gotMax != tt.wantMax {
				t.Errorf("MinMax(%v) = (%d, %d), want (%d, %d)", tt.in, gotMin, gotMax, tt.wantMin, tt.wantMax)
			}
		})
	}
}
