package kata

import "testing"

func TestRepeatText(t *testing.T) {
	tests := []struct {
		name  string
		text  string
		times int
		want  string
	}{
		{"zero times", "ab", 0, ""},
		{"negative times", "ab", -2, ""},
		{"once", "ab", 1, "ab"},
		{"multiple", "ab", 3, "ababab"},
		{"empty text", "", 5, ""},
		{"single char", "x", 4, "xxxx"},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			if got := RepeatText(tt.text, tt.times); got != tt.want {
				t.Errorf("RepeatText(%q, %d) = %q, want %q", tt.text, tt.times, got, tt.want)
			}
		})
	}
}
