package kata

import (
	"reflect"
	"testing"
)

func TestSafeSlice(t *testing.T) {
	values := []int{10, 20, 30, 40, 50}
	tests := []struct {
		name        string
		start, end  int
		want        []int
	}{
		{"normal", 1, 3, []int{20, 30}},
		{"start clamped", -5, 2, []int{10, 20}},
		{"end clamped", 3, 99, []int{40, 50}},
		{"both clamped", -10, 100, []int{10, 20, 30, 40, 50}},
		{"empty range", 2, 2, []int{}},
		{"start after end", 4, 1, []int{}},
		{"full", 0, 5, []int{10, 20, 30, 40, 50}},
	}
	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			got := SafeSlice(values, tt.start, tt.end)
			if !reflect.DeepEqual(got, tt.want) {
				t.Errorf("SafeSlice(%v, %d, %d) = %v, want %v", values, tt.start, tt.end, got, tt.want)
			}
		})
	}
}
