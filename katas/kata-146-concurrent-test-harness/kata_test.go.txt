package kata

import (
	"errors"
	"sync"
	"testing"
	"time"
)

func TestRunTestsCounts(t *testing.T) {
	tests := []TestCase{
		{Name: "ok1", Run: func() error { return nil }},
		{Name: "bad1", Run: func() error { return errors.New("x") }},
		{Name: "ok2", Run: func() error { return nil }},
	}
	passed, failed := RunTests(tests, 2)
	if passed != 2 || failed != 1 {
		t.Errorf("passed=%d failed=%d, want 2/1", passed, failed)
	}
}

func TestRunTestsEmpty(t *testing.T) {
	passed, failed := RunTests(nil, 4)
	if passed != 0 || failed != 0 {
		t.Errorf("passed=%d failed=%d, want 0/0", passed, failed)
	}
}

func TestRunTestsConcurrencyLimit(t *testing.T) {
	var mu sync.Mutex
	active := 0
	maxActive := 0
	var wg sync.WaitGroup
	tests := make([]TestCase, 6)
	for i := range tests {
		tests[i] = TestCase{
			Name: "t",
			Run: func() error {
				mu.Lock()
				active++
				if active > maxActive {
					maxActive = active
				}
				mu.Unlock()
				time.Sleep(30 * time.Millisecond)
				mu.Lock()
				active--
				mu.Unlock()
				wg.Done()
				return nil
			},
		}
	}
	wg.Add(6)
	RunTests(tests, 2)
	wg.Wait()
	if maxActive > 2 {
		t.Errorf("maxActive = %d, want <= 2", maxActive)
	}
}

func TestRunTestsUnlimited(t *testing.T) {
	var mu sync.Mutex
	active := 0
	maxActive := 0
	var wg sync.WaitGroup
	tests := make([]TestCase, 4)
	for i := range tests {
		tests[i] = TestCase{
			Name: "t",
			Run: func() error {
				mu.Lock()
				active++
				if active > maxActive {
					maxActive = active
				}
				mu.Unlock()
				time.Sleep(20 * time.Millisecond)
				mu.Lock()
				active--
				mu.Unlock()
				wg.Done()
				return nil
			},
		}
	}
	wg.Add(4)
	RunTests(tests, 0) // unlimited
	wg.Wait()
	if maxActive < 2 {
		t.Errorf("maxActive = %d, want > 1 with unlimited workers", maxActive)
	}
}
