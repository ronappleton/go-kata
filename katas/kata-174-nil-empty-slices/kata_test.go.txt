package kata

import (
	"testing"
)

func TestClassifySlice(t *testing.T) {
	var nilSlice []int
	if got := ClassifySlice(nilSlice); got != "nil" {
		t.Errorf("ClassifySlice(nil) = %q, want %q", got, "nil")
	}

	empty := []int{}
	if got := ClassifySlice(empty); got != "empty" {
		t.Errorf("ClassifySlice([]) = %q, want %q", got, "empty")
	}

	populated := []int{1, 2, 3}
	if got := ClassifySlice(populated); got != "populated" {
		t.Errorf("ClassifySlice([1,2,3]) = %q, want %q", got, "populated")
	}
}

func TestSafeMapGet(t *testing.T) {
	var nilMap map[string]string
	if _, ok := SafeMapGet(nilMap, "x"); ok {
		t.Error("SafeMapGet(nil, x) should return false")
	}

	m := map[string]string{"a": "1", "b": "2"}
	val, ok := SafeMapGet(m, "a")
	if !ok || val != "1" {
		t.Errorf("SafeMapGet(m, a) = (%q, %v), want ('1', true)", val, ok)
	}

	_, ok = SafeMapGet(m, "z")
	if ok {
		t.Error("SafeMapGet(m, z) should return false")
	}
}

func TestEnsureEmpty(t *testing.T) {
	var nilSlice []string
	got := EnsureEmpty(nilSlice)
	if got == nil {
		t.Error("EnsureEmpty(nil) returned nil, want non-nil empty slice")
	}
	if len(got) != 0 {
		t.Errorf("EnsureEmpty(nil) len = %d, want 0", len(got))
	}

	nonEmpty := []string{"a", "b"}
	got = EnsureEmpty(nonEmpty)
	if len(got) != 2 {
		t.Errorf("EnsureEmpty([a,b]) len = %d, want 2", len(got))
	}
	// Should be a copy
	got[0] = "x"
	if nonEmpty[0] == "x" {
		t.Error("EnsureEmpty should return a copy, not mutate input")
	}
}

func TestNormalizeMap(t *testing.T) {
	var nilMap map[string]string
	got := NormalizeMap(nilMap)
	if got == nil {
		t.Error("NormalizeMap(nil) returned nil, want non-nil empty map")
	}

	m := map[string]string{"a": "val", "b": "", "c": "x"}
	got = NormalizeMap(m)
	if got["a"] != "val" {
		t.Errorf("NormalizeMap()[a] = %q, want 'val'", got["a"])
	}
	if got["b"] != "default" {
		t.Errorf("NormalizeMap()[b] = %q, want 'default'", got["b"])
	}
	if got["c"] != "x" {
		t.Errorf("NormalizeMap()[c] = %q, want 'x'", got["c"])
	}
	// Should not mutate original
	if m["b"] != "" {
		t.Error("NormalizeMap should not mutate input")
	}
}
