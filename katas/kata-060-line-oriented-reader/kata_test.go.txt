package kata

import (
	"strings"
	"testing"
)

func TestReadLines(t *testing.T) {
	t.Run("reads lines", func(t *testing.T) {
		got, err := ReadLines(strings.NewReader("a\nb\nc\n"), 10)
		if err != nil {
			t.Fatal(err)
		}
		if len(got) != 3 || got[0] != "a" || got[2] != "c" {
			t.Errorf("ReadLines = %v", got)
		}
	})
	t.Run("last line without newline", func(t *testing.T) {
		got, err := ReadLines(strings.NewReader("a\nb"), 10)
		if err != nil {
			t.Fatal(err)
		}
		if len(got) != 2 || got[1] != "b" {
			t.Errorf("ReadLines = %v", got)
		}
	})
	t.Run("oversized line errors", func(t *testing.T) {
		if _, err := ReadLines(strings.NewReader("abcdef"), 3); err == nil {
			t.Error("expected error for line exceeding maxLine")
		}
	})
	t.Run("empty reader", func(t *testing.T) {
		got, err := ReadLines(strings.NewReader(""), 10)
		if err != nil {
			t.Fatal(err)
		}
		if len(got) != 0 {
			t.Errorf("ReadLines = %v, want empty", got)
		}
	})
}
