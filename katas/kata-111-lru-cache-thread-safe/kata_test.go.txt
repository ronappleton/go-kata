package kata

import (
	"sync"
	"testing"
)

func TestNewLRUCache(t *testing.T) {
	if _, err := NewLRUCache(0); err == nil {
		t.Error("expected error for capacity 0")
	}
	if _, err := NewLRUCache(-1); err == nil {
		t.Error("expected error for negative capacity")
	}
	c, err := NewLRUCache(2)
	if err != nil {
		t.Fatal(err)
	}
	if c.Len() != 0 {
		t.Errorf("Len = %d, want 0", c.Len())
	}
}

func TestLRUCacheGetPut(t *testing.T) {
	c, _ := NewLRUCache(3)
	if _, ok := c.Get("missing"); ok {
		t.Error("Get on missing key should return ok=false")
	}
	c.Put("a", "1")
	c.Put("b", "2")
	c.Put("c", "3")
	if v, ok := c.Get("a"); !ok || v != "1" {
		t.Errorf("Get(a) = %q,%v want 1,true", v, ok)
	}
	if c.Len() != 3 {
		t.Errorf("Len = %d, want 3", c.Len())
	}
	// update existing key keeps size
	c.Put("a", "10")
	if c.Len() != 3 {
		t.Errorf("Len after update = %d, want 3", c.Len())
	}
	if v, _ := c.Get("a"); v != "10" {
		t.Errorf("Get(a) after update = %q, want 10", v)
	}
}

func TestLRUCacheEviction(t *testing.T) {
	c, _ := NewLRUCache(2)
	c.Put("a", "1")
	c.Put("b", "2")
	// touch a so b becomes LRU
	c.Get("a")
	c.Put("c", "3") // evicts b
	if _, ok := c.Get("b"); ok {
		t.Error("b should have been evicted")
	}
	if _, ok := c.Get("a"); !ok {
		t.Error("a should still be present")
	}
	if _, ok := c.Get("c"); !ok {
		t.Error("c should be present")
	}
	if c.Len() != 2 {
		t.Errorf("Len = %d, want 2", c.Len())
	}
}

func TestLRUCacheConcurrent(t *testing.T) {
	c, _ := NewLRUCache(100)
	var wg sync.WaitGroup
	for i := 0; i < 8; i++ {
		wg.Add(1)
		go func(n int) {
			defer wg.Done()
			for j := 0; j < 200; j++ {
				k := string(rune('a' + (n*200+j)%26))
				c.Put(k, "v")
				c.Get(k)
				c.Len()
			}
		}(i)
	}
	wg.Wait()
}
