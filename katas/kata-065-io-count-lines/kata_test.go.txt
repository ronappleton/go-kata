package kata

import (
	"strings"
	"testing"
)

func TestCountLines(t *testing.T) {
	t.Run("nil reader errors", func(t *testing.T) {
		if _, err := CountLines(nil); err == nil {
			t.Error("expected error for nil reader")
		}
	})
	t.Run("counts lines", func(t *testing.T) {
		n, err := CountLines(strings.NewReader("a\nb\nc\n"))
		if err != nil {
			t.Fatal(err)
		}
		if n != 3 {
			t.Errorf("CountLines = %d, want 3", n)
		}
	})
	t.Run("empty input zero", func(t *testing.T) {
		n, err := CountLines(strings.NewReader(""))
		if err != nil {
			t.Fatal(err)
		}
		if n != 0 {
			t.Errorf("CountLines = %d, want 0", n)
		}
	})
	t.Run("last line without newline", func(t *testing.T) {
		n, _ := CountLines(strings.NewReader("a\nb"))
		if n != 2 {
			t.Errorf("CountLines = %d, want 2", n)
		}
	})
}
